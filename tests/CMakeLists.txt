cmake_minimum_required(VERSION 3.18)
project(wxshadow_tests LANGUAGES C CXX ASM)
enable_testing()
set(CMAKE_CXX_STANDARD 17)
set(DOBBY_WITH_WXSHADOW ON CACHE BOOL "" FORCE)
set(DOBBY_WXSHADOW_TESTING ON CACHE BOOL "" FORCE)
set(DOBBY_WXSHADOW_UAPI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../shared" CACHE PATH "" FORCE)
option(DOBBY_JNI_TESTS "Run JNI tests with a real host JVM" OFF)
if(DOBBY_JNI_TESTS)
  set(DOBBY_WITH_JNI ON CACHE BOOL "" FORCE)
  find_package(JNI REQUIRED)
  find_package(Java REQUIRED COMPONENTS Development)
endif()
add_subdirectory(../Dobby dobby)
if(DOBBY_JNI_TESTS)
  set(jni_classes "${CMAKE_CURRENT_BINARY_DIR}/jni-classes")
  add_custom_command(OUTPUT "${jni_classes}/DobbyJniFixture.class"
    COMMAND ${CMAKE_COMMAND} -E make_directory "${jni_classes}"
    COMMAND ${Java_JAVAC_EXECUTABLE} --release 8 -d "${jni_classes}" "${CMAKE_CURRENT_SOURCE_DIR}/jni/DobbyJniFixture.java"
    DEPENDS jni/DobbyJniFixture.java VERBATIM)
  add_custom_target(jni_fixture DEPENDS "${jni_classes}/DobbyJniFixture.class")
  foreach(kind static shared)
    add_executable(jni_${kind}_test jni/jni_tests.cc)
    add_dependencies(jni_${kind}_test jni_fixture)
    target_compile_options(jni_${kind}_test PRIVATE -UNDEBUG)
    target_include_directories(jni_${kind}_test PRIVATE ${JNI_INCLUDE_DIRS})
    if(kind STREQUAL "static")
      target_link_libraries(jni_${kind}_test PRIVATE dobby_static ${JAVA_JVM_LIBRARY} dl pthread)
    else()
      set_target_properties(jni_${kind}_test PROPERTIES ENABLE_EXPORTS ON)
      target_link_libraries(jni_${kind}_test PRIVATE dobby ${JAVA_JVM_LIBRARY} dl pthread)
    endif()
    add_test(NAME jni_${kind} COMMAND jni_${kind}_test "${jni_classes}")
  endforeach()
endif()
foreach(kind static shared)
  add_executable(wxshadow_${kind}_test wxshadow/host_tests.cc)
  target_compile_options(wxshadow_${kind}_test PRIVATE -UNDEBUG)
  target_include_directories(wxshadow_${kind}_test PRIVATE ../shared ../Dobby/include ../Dobby ../Dobby/source ../Dobby/source/dobby ../Dobby/external ../Dobby/external/logging)
  if(kind STREQUAL "static")
    target_compile_definitions(wxshadow_${kind}_test PRIVATE TEST_LINK_STATIC=1)
    target_link_libraries(wxshadow_${kind}_test PRIVATE dobby_static dl pthread)
  else()
    set_target_properties(wxshadow_${kind}_test PROPERTIES ENABLE_EXPORTS ON)
    target_link_libraries(wxshadow_${kind}_test PRIVATE dobby dl pthread)
  endif()
  add_test(NAME wxshadow_${kind} COMMAND wxshadow_${kind}_test)
endforeach()
add_executable(wxshadow_adapter_test wxshadow/adapter_tests.c ../kpms/wxshadow/linux/wxshadow_linux_adapter.c)
target_compile_definitions(wxshadow_adapter_test PRIVATE WXSHADOW_ADAPTER_TEST=1)
target_compile_options(wxshadow_adapter_test PRIVATE -UNDEBUG)
target_include_directories(wxshadow_adapter_test PRIVATE ../shared wxshadow)
add_test(NAME wxshadow_vm_transactions COMMAND wxshadow_adapter_test)
add_executable(wxshadow_protocol_test wxshadow/protocol_tests.c ../kpms/wxshadow/wxshadow_protocol.c)
target_compile_definitions(wxshadow_protocol_test PRIVATE WXSHADOW_PROTOCOL_TEST=1 WXSHADOW_NATIVE_ADAPTER=1)
target_compile_options(wxshadow_protocol_test PRIVATE -UNDEBUG)
target_include_directories(wxshadow_protocol_test PRIVATE ../shared wxshadow)
add_test(NAME wxshadow_protocol COMMAND wxshadow_protocol_test)
add_subdirectory(compat)
