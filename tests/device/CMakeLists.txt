cmake_minimum_required(VERSION 3.18)
project(wxshadow_device_tests LANGUAGES C CXX ASM)
if(NOT ANDROID OR NOT CMAKE_ANDROID_ARCH_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Device tests require Android arm64-v8a")
endif()
set(CMAKE_CXX_STANDARD 17)
set(DOBBY_WITH_WXSHADOW ON CACHE BOOL "" FORCE)
set(DOBBY_WXSHADOW_TESTING OFF CACHE BOOL "" FORCE)
set(DOBBY_WXSHADOW_UAPI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../shared" CACHE PATH "" FORCE)
add_subdirectory(../../Dobby dobby)
foreach(kind static shared)
  add_executable(wxshadow_device_${kind} device_tests.cc)
  target_compile_options(wxshadow_device_${kind} PRIVATE -mbranch-protection=none)
  target_include_directories(wxshadow_device_${kind} PRIVATE ../../Dobby/include ../../shared)
  if(kind STREQUAL "static")
    target_link_libraries(wxshadow_device_${kind} PRIVATE dobby_static dl)
  else()
    target_link_libraries(wxshadow_device_${kind} PRIVATE dobby dl)
  endif()
endforeach()
add_executable(wxshadow_protocol_client ../../kpms/wxshadow/wxshadow_protocol_client.c)
target_include_directories(wxshadow_protocol_client PRIVATE ../../shared)
